cmake_minimum_required(VERSION 3.16)
project(FastCopyMint VERSION 0.2.8 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(FASTCOPY_STATIC_ENGINE "Build a self-contained Linux CLI engine" OFF)
find_package(Threads REQUIRED)
add_library(upstream_xxhash STATIC external/xxhash/xxhash.c)
target_include_directories(upstream_xxhash PUBLIC external/xxhash)
add_executable(fastcopy-mint-cli linux/engine.cpp)
target_compile_options(fastcopy-mint-cli PRIVATE -Wall -Wextra -Wpedantic)
target_compile_definitions(fastcopy-mint-cli PRIVATE _FILE_OFFSET_BITS=64)
target_link_libraries(fastcopy-mint-cli PRIVATE upstream_xxhash Threads::Threads)
if(FASTCOPY_STATIC_ENGINE)
  target_link_options(fastcopy-mint-cli PRIVATE -static)
endif()
install(TARGETS fastcopy-mint-cli RUNTIME DESTINATION bin)
install(PROGRAMS linux/gui.py DESTINATION bin RENAME fastcopy)
# Remove only the compatibility symlink installed by the previous release.
# Do not remove an unrelated executable or a symlink with a different target.
install(CODE [[
  set(old_launcher "$ENV{DESTDIR}${CMAKE_INSTALL_PREFIX}/bin/fastcopy-mint")
  if(IS_SYMLINK "${old_launcher}")
    file(READ_SYMLINK "${old_launcher}" old_target)
    if(old_target STREQUAL "fastcopy")
      file(REMOVE "${old_launcher}")
    endif()
  endif()
]])
install(FILES linux/tray.py linux/rc_layout.json linux/fastcopy.png linux/menu.png linux/pause.png linux/play.png help/fastcopy_eng.htm DESTINATION share/fastcopy-mint)
install(FILES linux/fastcopy.png DESTINATION share/icons/hicolor/64x64/apps RENAME fastcopy-mint.png)
install(FILES packaging/fastcopy-mint.desktop DESTINATION share/applications)
install(FILES README.md doc/license-gpl3.txt doc/xxhash-LICENSE.txt DESTINATION share/doc/fastcopy-mint)
install(PROGRAMS packaging/selection.py DESTINATION bin RENAME fastcopy-mint-selection)
install(PROGRAMS packaging/integrate.sh DESTINATION bin RENAME fastcopy-mint-integrate)
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_test(NAME resource-layout COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_resources.py ${CMAKE_CURRENT_SOURCE_DIR})
  add_test(NAME theme-detection COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_theme.py ${CMAKE_CURRENT_SOURCE_DIR})
  add_test(NAME engine-regression COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_engine.py $<TARGET_FILE:fastcopy-mint-cli>)
endif()
